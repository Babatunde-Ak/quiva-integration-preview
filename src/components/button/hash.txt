import React, { useState } from "react";
import {
  HederaSessionEvent,
  HederaJsonRpcMethod,
  DAppConnector,
  HederaChainId,
} from "@hashgraph/hedera-wallet-connect";
import { LedgerId } from "@hiero-ledger/sdk";

const projectId = "YOUR_WALLETCONNECT_PROJECT_ID"; // from cloud.walletconnect.com

const metadata = {
  name: "Quiva DApp", // your app name
  description: "Hedera WalletConnect Example",
  url: window.location.origin, // must match your actual domain
  icons: ["https://avatars.githubusercontent.com/u/31002956"],
};

// Instantiate the connector
const dAppConnector = new DAppConnector(
  metadata,
  LedgerId.Testnet, // use Mainnet if you're going live
  projectId,
  Object.values(HederaJsonRpcMethod),
  [HederaSessionEvent.ChainChanged, HederaSessionEvent.AccountsChanged],
  [HederaChainId.Testnet, HederaChainId.Mainnet]
);

export default function HashConnect() {
  const [accountId, setAccountId] = useState<string | null>(null);

  const connectWallet = async () => {
    try {
      await dAppConnector.init({ logger: "error" });
      const { topic, pairingString } = await dAppConnector.openModal();

      // Wait for connection
      dAppConnector.on(HederaSessionEvent.AccountsChanged, (accounts) => {
        if (accounts.length > 0) {
          setAccountId(accounts[0]);
          console.log("Connected Account:", accounts[0]);
        }
      });

      console.log("Pairing string:", pairingString);
      console.log("Topic:", topic);
    } catch (error) {
      console.error("Error connecting to HashPack:", error);
    }
  };

  return (
    <div className="flex flex-col items-center p-4">
      <h2 className="text-xl font-semibold mb-4">HashPack Wallet Connect</h2>
      {accountId ? (
        <p className="text-green-600">✅ Connected: {accountId}</p>
      ) : (
        <button
          onClick={connectWallet}
          className="bg-indigo-600 text-white px-4 py-2 rounded-lg hover:bg-indigo-700 transition"
        >
          Connect HashPack
        </button>
      )}
    </div>
  );
}
